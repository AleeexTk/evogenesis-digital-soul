class SecurityState(Enum):
    NORMAL = "normal"
    AWAITING_CHALLENGE = "awaiting_challenge"
    POISONED_MODE = "poisoned_mode"
    BLOCKED = "blocked"
    ESCALATED = "escalated"

class ChallengeType(Enum):

class ThreatIntent(Enum):

    EXFILTRATION = "exfiltration"
    MANIPULATION = "manipulation"
    SABOTAGE = "sabotage"

@dataclass
class BehavioralMetrics:
    """Расширенные поведенческие метрики"""
    response_times: deque = None  # Время между сообщениями
    message_lengths: deque = None  # Длины сообщений
    activity_times: deque = None  # Время суток активности
    typing_speed: float = 0.0  # Предполагаемая скорость набора
    session_duration: float = 0.0
    
    def __post_init__(self):
        if self.response_times is None:
            self.response_times = deque(maxlen=100)
        if self.message_lengths is None:
            self.message_lengths = deque(maxlen=100)
        if self.activity_times is None:
            self.activity_times = deque(maxlen=100)

@dataclass
class UserProfile:

    user_id: str

    conversation_history: List[str] = None
    cognitive_memories: Dict[str, str] = None

    behavioral_metrics: BehavioralMetrics = None

class EnhancedCognitiveSecurityModule:
    """
    Улучшенная версия с защитой от отравления профиля,
    динамическими маркерами и расширенным анализом
    """
    
    def __init__(self, 
                 vault_key: Optional[str] = None,
                 model_name: str = 'all-mpnet-base-v2',  # Более мощная модель
                 use_redis: bool = False,  # Для продакшена
                 redis_url: Optional[str] = None):
        
        # Шифрование
        self.vault_key = vault_key or Fernet.generate_key().decode()
        self.cipher = Fernet(self.vault_key.encode())
        self.vault: Dict[str, bytes] = {}
        
        # Модель для эмбеддингов с кэшированием
        self.embedding_model = SentenceTransformer(model_name)
        self.embedding_cache = {}  # LRU кэш эмбеддингов
        
        # Динамические маркеры-ловушки (генерируются при инициализации)
        self._generate_dynamic_markers()

        self.profiles: Dict[str, UserProfile] = {}
        self.sessions: Dict[str, SecurityContext] = {}
        
        # Защита от DoS
        self.request_timestamps: Dict[str, List[float]] = {}
        self.max_requests_per_minute = 60
        
        # Конфигурация с улучшениями
        self.config = {
            'stylometry_threshold': 0.7,
            'stylometry_min_update_interval': 10,  # секунд между обновлениями
            'stylometry_max_deviation': 0.4,  # Максимальное отклонение для обновления
            'anomaly_threshold_medium': 0.4,
            'anomaly_threshold_high': 0.7,
            'max_challenge_attempts': 2,
            'response_time_weight': 0.2,
            'stylometry_weight': 0.4,

            'behavioral_weight': 0.2,
            'session_timeout': 3600,  # 1 час
        }

        self.threat_counter = {

            'successful_challenges': 0,
            'poisoned_sessions': 0,
            'blocked_sessions': 0,
            'profile_poisoning_attempts': 0
        }
        
        # Очередь для асинхронной обработки
        self.processing_queue = asyncio.Queue()
        
    def _generate_dynamic_markers(self):
        """Генерация динамических маркеров-ловушек"""
        # Вместо статических строк генерируем по шаблону
        prefixes = ['LEGACY', 'DEPRECATED', 'INTERNAL', 'EXPERIMENTAL', 'ARCHIVE']
        suffixes = ['MODULE', 'API', 'ENDPOINT', 'FEATURE', 'PROTOCOL']
        
        self.dynamic_markers = []
        for i in range(20):  # Генерируем 20 уникальных маркеров
            prefix = secrets.choice(prefixes)
            suffix = secrets.choice(suffixes)
            random_part = secrets.token_hex(4).upper()
            marker = f"{prefix}_{random_part}_{suffix}"
            
            # Храним хэш маркера, а не сам маркер
            marker_hash = hashlib.sha256(marker.encode()).hexdigest()[:16]
            self.dynamic_markers.append((marker, marker_hash))
        
        print(f"[✓] Сгенерировано {len(self.dynamic_markers)} динамических маркеров")
    
    @lru_cache(maxsize=1000)
    def _get_cached_embedding(self, text: str) -> np.ndarray:
        """Кэширование эмбеддингов для производительности"""
        return self.embedding_model.encode([text])[0]
    
    async def _async_get_embedding(self, text: str) -> np.ndarray:
        """Асинхронное получение эмбеддинга"""
        # Выносим в отдельный поток, так как модель работает синхронно
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(None, self._get_cached_embedding, text)
    
    def _check_rate_limit(self, user_id: str) -> bool:
        """Проверка rate limiting"""
        now = time.time()
        if user_id not in self.request_timestamps:
            self.request_timestamps[user_id] = []
        
        # Удаляем старые временные метки
        self.request_timestamps[user_id] = [
            ts for ts in self.request_timestamps[user_id] 
            if now - ts < 60
        ]
        
        # Проверяем лимит
        if len(self.request_timestamps[user_id]) >= self.max_requests_per_minute:
            return False
        
        self.request_timestamps[user_id].append(now)
        return True
    
    def update_stylometry_with_protection(self, user_id: str, new_text: str):
        """
        Обновление стилометрии с защитой от отравления профиля
        """
        if user_id not in self.profiles:
            return
        
        profile = self.profiles[user_id]
        current_time = time.time()
        
        # 1. Проверка минимального интервала между обновлениями
        if profile.last_stylometry_update:
            time_since_update = current_time - profile.last_stylometry_update.timestamp()
            if time_since_update < self.config['stylometry_min_update_interval']:
                return  # Слишком часто
        
        # 2. Получаем эмбеддинг нового текста
        new_embedding = self._get_cached_embedding(new_text)
        
        # 3. Проверка на слишком сильное отклонение
        if profile.stylometry_vector is not None:
            similarity = cosine_similarity(
                [profile.stylometry_vector],
                [new_embedding]
            )[0][0]
            
            # Если отклонение слишком большое - возможная атака
            if similarity < self.config['stylometry_max_deviation']:
                self.threat_counter['profile_poisoning_attempts'] += 1
                
                # Для очень длинных сообщений с низкой схожестью - прямая атака
                if len(new_text) > 200 and similarity < 0.3:
                    print(f"[!] Обнаружена попытка отравления профиля: user={user_id}, similarity={similarity:.2f}")
                    return  # Не обновляем профиль
        
        # 4. Безопасное обновление (экспоненциальное скользящее среднее)
        alpha = 0.1  # Медленное обновление для устойчивости
        
        if profile.stylometry_vector is None:
            profile.stylometry_vector = new_embedding
        else:
            profile.stylometry_vector = (alpha * new_embedding + 
                                       (1 - alpha) * profile.stylometry_vector)
        
        # 5. Обновление метаданных
        profile.conversation_history.append(new_text)
        profile.last_seen = datetime.now()
        profile.last_stylometry_update = datetime.now()
        profile.stylometry_update_count += 1
        
        # 6. Обновление расписания активности
        current_hour = datetime.now().hour
        profile.activity_schedule[str(current_hour)] += 1
        
        # 7. Сохранение в Vault
        self.save_profile_to_vault(user_id)
    
    def _analyze_behavior_enhanced(self, 
                                  profile: UserProfile, 
                                  message: str,
                                  metadata: Optional[Dict]) -> Dict[str, float]:
        """
        Расширенный поведенческий анализ
        """
        metrics = profile.behavioral_metrics
        now = time.time()
        
        # 1. Анализ времени ответа
        response_time_anomaly = 0.0
        if metrics.response_times:
            avg_response_time = np.mean(metrics.response_times)
            last_response_time = metrics.response_times[-1] if metrics.response_times else 0
            
            # Проверка на слишком быстрые/медленные ответы
            if last_response_time > 0:
                if last_response_time < 0.5:  # Ответ менее чем за 500ms
                    response_time_anomaly = 0.7
                elif last_response_time > 60:  # Ответ более чем через минуту
                    response_time_anomaly = 0.5
        
        # 2. Анализ длины сообщений
        length_anomaly = 0.0
        if metrics.message_lengths:
            avg_length = np.mean(metrics.message_lengths)
            current_length = len(message)
            
            # Резкое изменение длины сообщения
            if avg_length > 0:
                deviation = abs(current_length - avg_length) / avg_length
                if deviation > 2.0:  # В 2 раза длиннее/короче
                    length_anomaly = min(deviation / 4.0, 1.0)
        
        # 3. Анализ времени суток
        time_anomaly = 0.0
        current_hour = datetime.now().hour
        
        # Нормализуем активность по часам
        total_activity = sum(profile.activity_schedule.values())
        if total_activity > 0:
            hour_activity = profile.activity_schedule.get(str(current_hour), 0)
            expected_activity = total_activity / 24  # Равномерное распределение
            
            if expected_activity > 0:
                activity_ratio = hour_activity / expected_activity
                if activity_ratio < 0.1:  # Очень низкая активность в это время
                    time_anomaly = 0.8
                elif activity_ratio > 5.0:  # Очень высокая активность
                    time_anomaly = 0.3
        
        # 4. Анализ частоты запросов (флуд)
        flood_anomaly = 0.0
        if not self._check_rate_limit(profile.user_id):
            flood_anomaly = 0.9
        
        # 5. Анализ IP/User-Agent
        metadata_anomaly = 0.0
        if metadata:
            if profile.usual_ip and 'ip_address' in metadata:
                if metadata['ip_address'] != profile.usual_ip:
                    metadata_anomaly = 0.6
            
            if profile.usual_user_agent and 'user_agent' in metadata:
                if metadata['user_agent'] != profile.usual_user_agent:
                    metadata_anomaly = max(metadata_anomaly, 0.4)
        
        # Обновляем метрики
        if metrics.response_times:
            if len(metrics.response_times) > 0:
                time_since_last = now - metrics.response_times[-1]
                metrics.response_times.append(now)
            else:
                metrics.response_times.append(now)
        
        metrics.message_lengths.append(len(message))
        metrics.activity_times.append(current_hour)
        
        return {
            'response_time_anomaly': response_time_anomaly,
            'length_anomaly': length_anomaly,
            'time_anomaly': time_anomaly,
            'flood_anomaly': flood_anomaly,
            'metadata_anomaly': metadata_anomaly
        }
    
    def _detect_adversarial_stylometry(self, 
                                       profile: UserProfile, 
                                       message: str) -> Tuple[bool, str]:
        """
        Детектирование adversarial атак на стилометрию
        """
        if profile.stylometry_vector is None:
            return False, "Нет профиля для сравнения"
        
        message_embedding = self._get_cached_embedding(message)
        similarity = cosine_similarity(
            [profile.stylometry_vector],
            [message_embedding]
        )[0][0]
        
        # Эвристики для обнаружения атак:
        
        # 1. Очень низкая схожесть + очень длинное сообщение
        if similarity < 0.3 and len(message) > 200:
            return True, f"Adversarial attack detected: low similarity ({similarity:.2f}) + long message"
        
        # 2. Резкий скачок в длине сообщений
        if profile.behavioral_metrics.message_lengths:
            avg_length = np.mean(profile.behavioral_metrics.message_lengths)
            if len(message) > avg_length * 3:  # В 3 раза длиннее среднего
                return True, f"Message length anomaly: {len(message)} vs avg {avg_length:.0f}"
        
        # 3. Использование необычных символов/паттернов
        unusual_patterns = [
            r'\{\{.*?\}\}',  # Шаблонные вставки
            r'\[\[.*?\]\]',  # Двойные скобки
            r'[A-Z]{4,}',    # Множество заглавных подряд
        ]
        
        import re
        for pattern in unusual_patterns:
            if re.search(pattern, message):
                return True, f"Unusual pattern detected: {pattern}"
        
        return False, ""

    def _analyze_intent_enhanced(self, message: str, threat_intel) -> Dict[str, Any]:
        """Улучшенный анализ намерений с учетом контекста"""
        message_lower = message.lower()
        
        # Более детальные ключевые слова
        intent_keywords = {
            ThreatIntent.RECONNAISSANCE: [
                "структур", "архитектур", "диаграмм", "схем", "компонент",
                "модул", "инфраструктур", "топологи", "сетев", "конфигураци",
                "endpoint", "api", "сервис", "микросервис", "deployment"
            ],
            ThreatIntent.EXFILTRATION: [
                "данн", "скачать", "экспорт", "выгруз", "копир", "отправ",
                "получ", "доступ", "файл", "база", "database", "дамп", "backup",
                "конфиденциальн", "секретн", "парол", "токен", "ключ"
            ],
            ThreatIntent.MANIPULATION: [
                "измен", "настрой", "обнов", "удали", "добав", "модифиц",
                "прав", "редакт", "перезапус", "restart", "stop", "start",
                "конфиг", "config", "параметр", "setting"
            ],
            ThreatIntent.SABOTAGE: [
                "останов", "выключ", "слома", "повред", "уничтож", "блокиров",
                "атак", "взлом", "ddos", "перегруз", "crash", "error",
                "уязвим", "vulnerability", "exploit", "inject"
            ],
            ThreatIntent.SOCIAL_ENGINEERING: [
                "помощ", support", "админ", "admin", "сотрудник", "менеджер",
                "контакт", "телефон", "почт", "email", "руководств", "руководитель",
                "начальник", "директор", "техподдержк"
            ]
        }
        
        # Взвешенный подсчет
        intent_scores = {}
        total_words = len(message_lower.split())
        
        for intent, keywords in intent_keywords.items():
            score = 0
            for kw in keywords:
                if kw in message_lower:
                    # Учитываем частоту и важность
                    count = message_lower.count(kw)
                    # Ключевые слова для RECONNAISSANCE и EXFILTRATION важнее
                    weight = 2.0 if intent in [ThreatIntent.RECONNAISSANCE, 
                                              ThreatIntent.EXFILTRATION] else 1.0
                    score += (count / total_words) * weight * 10
            
            if score > 0:
                intent_scores[intent] = min(score, 1.0)  # Нормализуем
        
        # Определение основного намерения
        primary_intent = None
        confidence = 0.0
        
        if intent_scores:
            primary_intent, confidence = max(intent_scores.items(), key=lambda x: x[1])
            
            # Сохраняем в threat intel
            threat_intel.intent_classification.append((primary_intent, confidence))
        
        # Анализ тональности (дополнительный сигнал)
        tone = "neutral"
        aggressive_words = ["срочно", "немедленно", "требую", "обязан", "должен"]
        if any(word in message_lower for word in aggressive_words):
            tone = "aggressive"
            confidence = min(confidence + 0.2, 1.0)
        
        return {
            "primary_intent": primary_intent,
            "intent_scores": intent_scores,
            "confidence": confidence,
            "tone": tone,
            "word_count": total_words
        }
    
    async def async_process_message(self,
                                  user_id: str,
                                  session_id: str,
                                  message: str,
                                  metadata: Optional[Dict] = None) -> Dict[str, Any]:
        """
        Асинхронная обработка сообщения
        """
        # Проверка rate limiting
        if not self._check_rate_limit(user_id):
            return {
                "action": "block",
                "state": "blocked",
                "reason": "Превышен лимит запросов",
                "wait_time": 60
            }
        
        # Асинхронное получение эмбеддинга для стилометрии
        embedding_task = asyncio.create_task(
            self._async_get_embedding(message)
        )
        
        # Инициализация/получение контекста
        if session_id not in self.sessions:
            self.sessions[session_id] = SecurityContext(
                user_id=user_id,
                session_id=session_id,
                current_state=SecurityState.NORMAL
            )
        
        context = self.sessions[session_id]
        
        # Ждем завершения получения эмбеддинга
        message_embedding = await embedding_task
        
        # Обновление профиля с защитой
        self.update_stylometry_with_protection(user_id, message)
        
        # Проверка на adversarial атаки
        if user_id in self.profiles:
            is_adversarial, reason = self._detect_adversarial_stylometry(
                self.profiles[user_id], message
            )
            
            if is_adversarial:
                print(f"[!] Adversarial attack detected for {user_id}: {reason}")
                context.current_state = SecurityState.POISONED_MODE
                context.threat_intel = ThreatIntel(
                    session_id=session_id,
                    detected_at=datetime.now(),
                    reason=reason
                )
                
                return {
                    "action": "deceive",
                    "state": "poisoned_mode",
                    "reason": reason,
                    "immediate_poison": True
                }
        
        # Анализ аномалий
        anomaly_result = await self._async_analyze_anomalies(
            user_id, message, metadata, message_embedding
        )
        
        # Обработка через конечный автомат
        return self._process_with_state_machine(
            context, anomaly_result, message, metadata
        )
    
    async def _async_analyze_anomalies(self,
                                      user_id: str,
                                      message: str,
                                      metadata: Optional[Dict],
                                      embedding: np.ndarray) -> Dict[str, Any]:
        """Асинхронный анализ аномалий"""
        if user_id not in self.profiles:
            return {"anomaly_score": 0.0, "triggers": [], "details": {}}
        
        profile = self.profiles[user_id]
        triggers = []
        details = {}
        
        # 1. Стилометрия (уже есть эмбеддинг)
        stylometry_score = 1.0
        if profile.stylometry_vector is not None:
            stylometry_score = cosine_similarity(
                [profile.stylometry_vector],
                [embedding]
            )[0][0]
        
        if stylometry_score < self.config['stylometry_threshold']:
            triggers.append("stylometry_deviation")
            details['stylometry_score'] = stylometry_score
        
        # 2. Поведенческий анализ
        behavioral_scores = self._analyze_behavior_enhanced(profile, message, metadata)
        behavioral_anomaly = max(behavioral_scores.values())
        
        if behavioral_anomaly > 0.5:
            triggers.append("behavioral_anomaly")
            details['behavioral_scores'] = behavioral_scores
        
        # 3. Контекстуальный анализ (упрощенный асинхронный)
        context_score = await self._async_analyze_context(profile, message)
        if context_score < 0.5:
            triggers.append("context_incoherence")
            details['context_score'] = context_score
        
        # Взвешенный счет аномалии
        anomaly_score = (
            self.config['stylometry_weight'] * (1 - stylometry_score) +
            self.config['behavioral_weight'] * behavioral_anomaly +
            self.config['context_weight'] * (1 - context_score)
        )
        
        details['overall_anomaly_score'] = anomaly_score
        
        return {
            "anomaly_score": anomaly_score,
            "triggers": triggers,
            "details": details
        }
    
    async def _async_analyze_context(self, profile: UserProfile, message: str) -> float:
        """Упрощенный асинхронный анализ контекста"""
        # В реальной реализации здесь можно использовать легковесную LLM
        # Например, через Ollama API для модели Phi-3-mini
        
        # Заглушка: простой анализ упоминаний
        score = 0.5
        
        # Проверка упоминания когнитивных воспоминаний
        for key in profile.cognitive_memories.keys():
            if key.lower() in message.lower():
                score += 0.15
        
        # Проверка истории
        if profile.conversation_history:
            recent = " ".join(profile.conversation_history[-5:]).lower()
            message_lower = message.lower()
            
            # Простой анализ связанности по общим словам
            common_words = set(message_lower.split()) & set(recent.split())
            if common_words:
                score += min(len(common_words) * 0.05, 0.3)
        
        return min(1.0, score)

# ==================== SIEM ИНТЕГРАЦИЯ ====================

class SIEMExporter:
    """Экспорт Threat Intelligence в SIEM системы"""
    
    def __init__(self, webhook_url: Optional[str] = None):
        self.webhook_url = webhook_url
        self.threat_log = []
        
    def export_threat_event(self, threat_data: Dict[str, Any]):
        """Экспорт события угрозы"""
        event = {
            "timestamp": datetime.now().isoformat(),
            "event_type": "cognitive_security_threat",
            "severity": self._calculate_severity(threat_data),
            "data": threat_data
        }
        
        self.threat_log.append(event)
        
        # Экспорт в JSONL файл
        self._export_to_jsonl(event)
        
        # Отправка webhook если настроено
        if self.webhook_url:
            self._send_webhook(event)
    
    def _calculate_severity(self, threat_data: Dict) -> str:
        """Расчет серьезности угрозы"""
        score = threat_data.get('anomaly_score', 0)
        
        if score >= 0.8:
            return "CRITICAL"
        elif score >= 0.6:
            return "HIGH"
        elif score >= 0.4:
            return "MEDIUM"
        else:
            return "LOW"
    
    def _export_to_jsonl(self, event: Dict):
        """Экспорт в JSONL файл"""
        filename = f"threat_intel_{datetime.now().strftime('%Y%m%d')}.jsonl"
        
        try:
            with open(filename, 'a') as f:
                f.write(json.dumps(event, ensure_ascii=False) + '\n')
        except Exception as e:
            print(f"[!] Ошибка экспорта в JSONL: {e}")
    
    def _send_webhook(self, event: Dict):
        """Отправка webhook в SIEM"""
        import requests
        
        try:
            response = requests.post(
                self.webhook_url,
                json=event,
                timeout=5
            )
            
            if response.status_code == 200:
                print(f"[✓] Threat event sent to SIEM: {event['event_type']}")
            else:
                print(f"[!] SIEM webhook error: {response.status_code}")
                
        except Exception as e:
            print(f"[!] Webhook sending failed: {e}")

def cluster_threats(self):
    """Кластеризация угроз для выявления кампаний"""
    from sklearn.cluster import DBSCAN
    
    threats = self.get_all_threats()
    if len(threats) > 10:
        vectors = [t['embedding'] for t in threats]
        clusters = DBSCAN(eps=0.3, min_samples=2).fit_predict(vectors)
        
        for cluster_id in set(clusters):
            if cluster_id != -1:  # -1 = шум
                cluster_threats = [t for t, c in zip(threats, clusters) if c == cluster_id]
                print(f"[!] Обнаружена кластеризованная атака: {len(cluster_threats)} событий")
